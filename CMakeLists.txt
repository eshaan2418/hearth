cmake_minimum_required(VERSION 3.24)
project(hearth VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(HEARTH_SANITIZE "" CACHE STRING "address, thread, or empty")
set(W -Wall -Wextra -Wpedantic -Werror -Wshadow -Wno-sign-compare)
if(HEARTH_SANITIZE STREQUAL "address")
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=all)
  add_link_options(-fsanitize=address,undefined)
elseif(HEARTH_SANITIZE STREQUAL "thread")
  add_compile_options(-fsanitize=thread)
  add_link_options(-fsanitize=thread)
endif()
find_package(Threads REQUIRED)
add_library(hearth STATIC src/http.cpp src/server.cpp)
target_include_directories(hearth PUBLIC include)
target_compile_options(hearth PRIVATE ${W})
target_link_libraries(hearth PUBLIC Threads::Threads)

add_executable(hearthd tools/hearthd.cpp)
target_compile_options(hearthd PRIVATE ${W})
target_link_libraries(hearthd PRIVATE hearth)
add_executable(hearth-load tools/load.cpp)
target_compile_options(hearth-load PRIVATE ${W})
target_link_libraries(hearth-load PRIVATE Threads::Threads)

find_package(GTest CONFIG QUIET)
if(NOT GTest_FOUND)
  include(FetchContent)
  FetchContent_Declare(googletest URL https://github.com/google/googletest/archive/refs/tags/v1.15.2.tar.gz
                       URL_HASH SHA256=7b42b4d6ed48810c5362c265a17faebe90dc2373c885e5216439d37927f02926)
  FetchContent_MakeAvailable(googletest)
endif()
enable_testing()
add_executable(hearth_tests tests/test_http.cpp tests/test_server.cpp)
target_compile_options(hearth_tests PRIVATE ${W})
target_link_libraries(hearth_tests PRIVATE hearth GTest::gtest_main)
include(GoogleTest)
gtest_discover_tests(hearth_tests DISCOVERY_TIMEOUT 60)
